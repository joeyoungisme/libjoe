#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <stddef.h>
#include <stdint.h>
#include <string.h>
#include <errno.h>



#include "misc.h"

static uint8_t* string_to_raw_data(const char *str, size_t *out_length)
{
    if (!str || !out_length) {
        return NULL;
    }
    
    size_t len = strlen(str);
    uint8_t *raw_data = (uint8_t*)malloc(len);
    if (!raw_data) {
        return NULL;
    }
    
    memcpy(raw_data, str, len);
    *out_length = len;
    return raw_data;
}

int main(int argc, char *argv[])
{
    if (argc < 2) {
        PRINT_DEBUG("%s invalid args.\n", argv[0]);
        PRINT_DEBUG("%s [execute command]\n", argv[0]);
        exit(EXIT_FAILURE);
    }

    char cmd[1024] = {0};
    int cnt = 0;
    for (int idx = 1; idx < argc; ++idx) {
        cnt += snprintf(cmd + cnt, 1024 - cnt, "%s ", argv[idx]);
    }
    PRINT_DEBUG("exe_cmd : %s\n", cmd);
    int res = exe_cmd(cmd);
    PRINT_DEBUG("result : %d , errno : %d.\n", res, errno);
    
    // Example usage of string_to_raw_data function
    size_t raw_len;
    uint8_t *raw_data = string_to_raw_data("test", &raw_len);
    if (raw_data) {
        PRINT_DEBUG("Converted string to raw data, length: %zu\n", raw_len);
        free(raw_data);
    }
    
    exit(EXIT_SUCCESS);
}
